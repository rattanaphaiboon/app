ผู้ใช้ในองค์กร · วันหยุดบริษัท

เข้าสู่ระบบด้วยอีเมลบริษัท แล้ว Rattana CI ดึงข้อมูลจากชีท Rattana Users for apps ให้เอง (ชื่อ รหัสพนักงาน คลัง W อีเมล คลังส่ง User Role LINE userId รูป) — โหลดมาเฉพาะแถวของคนที่เข้าสู่ระบบ

เข้าสู่ระบบ (ใส่ attribute เดียว)

<!-- ทั้งแอปต้องเข้าสู่ระบบก่อน · data-rt-login-role = User Role ขั้นต่ำ (ไม่ใส่ = ทุกคนในชีท) -->
<body class="rt-app" data-rt-login="ชื่อแอป" data-rt-login-role="3">
<button data-rt-user></button>                 <!-- ปุ่มรูป+ชื่อมุมขวาบน กดแล้วดูบัญชี/ออกจากระบบ -->

RattanaUI.user.ready.then(u => {
  u.name, u.nickname, u.empId, u.w, u.email, u.shipW, u.role, u.lineUserId, u.photo
  startApp(u);
});
RattanaUI.user.can(5);                           // User Role 5 ขึ้นไปไหม
RattanaUI.user.signOut();

วันหยุดบริษัท (ชีท วันหยุดนขัต)

RattanaUI.holidays.list().then(list => ...);      // [{ date: '2026-12-05', name: 'วันพ่อแห่งชาติ' }, …]
RattanaUI.holidays.is(new Date());                // ชื่อวันหยุด หรือ '' (หลังโหลด list แล้ว)
RattanaUI.holidays.next();                        // วันหยุดถัดไป
RattanaUI.holidays.addWorkdays(new Date(), 2);    // อีก 2 วันทำการ (ข้ามอาทิตย์+วันหยุด)

แจ้งเตือน · ค้นหา · สแกน

กระดิ่งแจ้งเตือนที่ตัวเลขไม่โดนตัด, ช่องค้นหาที่มีปุ่ม X ล้างทั้งหมด, ปุ่มสแกนบาร์โค้ด/QR สำหรับหน้าที่เกี่ยวกับสินค้า และปุ่มกลับขึ้นบนสุดที่ขึ้นเองทุกหน้า

กระดิ่งแจ้งเตือน + แจ้งเตือนนอกแอป

กดปุ่มทอง แล้วสลับไปแท็บ/แอปอื่นภายใน 5 วินาที — การแจ้งเตือนจะเด้งที่มุมจอ (มือถือ: แถบแจ้งเตือน) ตัวเลขบนไอคอนแอปก็เปลี่ยนด้วย

<button class="rt-btn rt-btn-glass rt-btn-icon rt-bell" data-rt-bell aria-label="การแจ้งเตือน">
  …ไอคอนกระดิ่ง…<span class="rt-count">3</span>
</button>

RattanaUI.setBadge(5);                      // ตัวเลขบนกระดิ่ง + ไอคอนแอป + ชื่อแท็บ (5)
RattanaUI.notifyPermission();               // ขออนุญาต (เรียกตอนผู้ใช้กดปุ่ม)
RattanaUI.notify('มีคำขออนุมัติใหม่', { body: 'REQ-00142 · ฿85,352', url: location.href, tag: 'req' });

ป๊อปอัปพรีเมียม

บนคอมและแล็ปท็อป เป็นหน้าต่างกลางจอที่ขยายออกจากจุดที่กด · บนมือถือ เปลี่ยนเป็นแผ่นเลื่อนขึ้นจากด้านล่างเอง ปัดลงเพื่อปิด และหลบคีย์บอร์ดให้อัตโนมัติ

โค้ด popup

// popup ทั่วไป: คอม = หน้าต่างกลางจอ · มือถือ = bottom sheet ปัดลงปิด
RattanaUI.popup({
  icon: 'question', title: 'บันทึกรายการนี้?', message: 'ตรวจข้อมูลก่อนกดบันทึก',
  content: '<div class="rt-field">...</div>',
  actions: [
    { label: 'ยกเลิก', variant: 'glass', value: false },
    { label: 'บันทึก', variant: 'gold', value: true, onClick: () => saveToServer() }   // คืน Promise = ปุ่มหมุนรอเอง
  ]
}).then(v => { ... });

RattanaUI.alert({ type: 'success', title: 'บันทึกเรียบร้อย' });         // success | error | warning | info
RattanaUI.menu(button, [{ label: 'แก้ไข', value: 'edit' }, '-', { label: 'ลบ', value: 'del', danger: true }]);
RattanaUI.drawer({ title: 'ตัวกรอง', content: '...', side: 'right' });
RattanaUI.toast('ลบแล้ว', { action: { label: 'เลิกทำ', onClick: undo } });

ใช้ได้ทั้ง iOS · Android · Laptop

Rattana CI ตรวจอุปกรณ์เองแล้วปรับให้เหมาะ ไม่ต้องเขียนเพิ่ม

อุปกรณ์ที่คุณใช้อยู่ตอนนี้—
รอยบาก / Dynamic Island / แถบ Homeเว้นขอบปลอดภัยให้เอง เมนูล่างไม่ทับแถบ home
ไม่ซูมเองตอนแตะช่องกรอก (iPhone)ช่องกรอก 16px บนจอสัมผัส ปุ่มกดง่ายอย่างน้อย 44px
Popup เป็น bottom sheet บนมือถือปัดลงปิด และลอยหลบคีย์บอร์ดเอง
สั่นตอบสนองบน Androidแตะ ยืนยัน หรือผิดพลาด จะสั่นเบา ๆ ต่างกัน
เมนูข้างพับเป็นเมนูสไลด์ใช้ .rt-shell: แล็ปท็อปเห็นเมนูข้าง มือถือ/แท็บเล็ตกดปุ่มเมนู
ติดตั้งเป็นแอปบนหน้าจอได้ใส่ meta ตามโค้ดเริ่มต้น แถบสถานะเปลี่ยนสีตามธีม

กฎที่ทุกแอปต้องทำตาม

วันที่อยู่ซ้ายบนทุกหน้า/ทุกส่วนที่มีวันที่ วางไว้ก่อนหัวข้อ ห้ามไว้ขวาบน
เขียว = อนุมัติ · แดง = ไม่อนุมัติปุ่มทองใช้กับงานทั่วไป ไม่ใช้แทนปุ่มยืนยัน
สถานะมีคำ + ไอคอนเสมอไม่ใช้สีอย่างเดียวในการสื่อความหมาย
ข้อมูล 2 บรรทัดข้อมูลหลักบน รายละเอียดเล็กล่าง แทนการเพิ่มคอลัมน์
ติดป้ายข้อมูลตัวอย่างข้อมูลสมมติต้องมีป้าย “ข้อมูลตัวอย่าง” ทุกครั้ง
โลโก้ใช้ไฟล์จริงเท่านั้นห้ามวาดใหม่ เปลี่ยนสี ยืด หรือใส่เอฟเฟกต์
